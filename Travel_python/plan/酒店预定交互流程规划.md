### 2.2 interaction 状态说明

| type | status | 含义 |
|------|--------|------|
| `hotel_select` | `waiting_user_hotel` | 等待用户选择酒店 |
| `hotel_select` | `waiting_user_decision` | researcher 发现全部没房，等待用户决定 |
| `hotel_rebooking` | `waiting_user_alternatives` | confirm 时原酒店没房，有替代酒店 |
| `hotel_rebooking` | `waiting_user_alarm` | confirm 时原酒店没房，无替代，问是否设闹钟 |
| `hotel_select` / `hotel_alarm` | `confirmed` | 用户已做选择/决定，继续后续流程 |
| 任意 | `cancelled` | 用户取消 |

```python
    interaction = state.get("interaction", {})
```

## 四、Supervisor 完整路由

```python
def supervisor_node(state):
    # 1. 已完成 → parse_plan
    if state.get("final_plan"):
        return {"next": "parse_plan"}

    # 2. 检查 interaction 等待状态
    interaction = state.get("interaction", {})
    if interaction.get("status", "").startswith("waiting_"):
        return {"next": "user_interaction"}

    # 3. 检查 needs_replan
    if state.get("needs_replan"):
        return {
            "next": "task_analyzer",
            "research_results": None,
            "draft_plan": None,
            "interaction": None,
            "task": None,
        }

    # 4. 各节点按序检查，跳过已有结果
    if "task" not in state:
        return {"next": "task_analyzer"}

    elif "research_results" not in state:
        return {"next": "researcher"}

    elif interaction.get("type") == "hotel_select" and not interaction.get("chosen"):
        # 有酒店需求但还没选 → user_interaction
        return {"next": "user_interaction"}

    elif "draft_plan" not in state:
        return {"next": "planner"}

    elif "scores" not in state or len(state.get("scores", [])) < state.get("max_iterations", 3):
        return {"next": "critic"}

    elif interaction.get("chosen") and state.get("booking_status") != "confirmed":
        # 用户选了酒店且未确认预定 → confirm_and_book
        return {"next": "confirm_and_book"}

    else:
        return {"next": "final_optimizer"}
```

## 六、User_Interaction 统一 Handler（完整流程）

### 6.1 定位与角色

```
┌─────────────────────────────────────────────────────┐
│              Supervisor 调度中心                    │
│                                                     │
│   state 检查 → interaction.status = waiting         │
│              ↓                                      │
│   router = "user_interaction"                       │
│              ↓                                      │
│   ┌─────────────────────────────────────────────┐  │
│   │         user_interaction_node                 │  │
│   │   统一入口：根据 interaction.status           │  │
│   │   调用对应 handler 处理用户回复               │  │
│   └─────────────────────────────────────────────┘  │
│              ↓                                      │
│         返回更新后的 state                           │
│              ↓                                      │
│   Supervisor 再次调度                               │
└─────────────────────────────────────────────────────┘
```

**核心职责**：接收用户回复 → 解析用户意图 → 更新 interaction 字段 → 返回新 state

### 6.2 Handler 入口

```python
def user_interaction_node(state):
    """
    统一入口，不分具体类型
    根据 interaction.status 决定调用哪个 handler
    """
    interaction = state.get("interaction", {})
    user_msg = state["messages"][-1].content
    status = interaction.get("status", "")

    # 根据状态分发到不同 handler
    if status == "waiting_user_hotel":
        return handle_hotel_selection(state, user_msg, interaction)

    elif status == "waiting_user_decision":
        return handle_no_hotel_decision(state, user_msg, interaction)

    elif status == "waiting_user_alternatives":
        return handle_hotel_alternatives(state, user_msg, interaction)

    elif status == "waiting_user_alarm":
        return handle_hotel_alarm_decision(state, user_msg, interaction)

    else:
        # 未知状态，不处理
        return {}
```

### 6.3 统一工具函数

#### parse_hotel_choice：解析用户选择的酒店

```python
def parse_hotel_choice(user_msg: str, hotels: list) -> dict | None:
    """
    从用户消息中解析出用户选择的酒店
    返回 Hotel 对象或 None
    """
    user_msg_clean = user_msg.strip()

    # ── 情况1：按序号选 ──
    if user_msg_clean.isdigit():
        index = int(user_msg_clean) - 1
        if 0 <= index < len(hotels):
            return hotels[index]

    # ── 情况2：按名称模糊匹配 ──
    for hotel in hotels:
        if hotel.get("hotel_name", "") in user_msg_clean:
            return hotel

    # ── 情况3：序号词匹配 ──
    order_map = {
        "第一": 0, "第二个": 1, "第三个": 2, "第四个": 3,
        "首家": 0, "第二": 1, "三": 2,
        "first": 0, "second": 1, "third": 2,
    }
    for keyword, idx in order_map.items():
        if keyword in user_msg_clean and idx < len(hotels):
            return hotels[idx]

    return None
```

#### build_*_prompt：构建前端提示

```python
def build_hotel_selection_prompt(hotels: list) -> dict:
    """用户选择酒店时的提示"""
    hotel_list = "\n".join([
        f"{i+1}. {h['hotel_name']} - ¥{h.get('price', '?')}/晚"
        for i, h in enumerate(hotels)
    ])
    return {
        "interaction": {
            "message": f"请从以下酒店中选择：\n{hotel_list}\n\n请回复序号或酒店名称"
        }
    }


def build_no_hotel_prompt() -> dict:
    """全部没房时的提示"""
    return {
        "interaction": {
            "message": "无法识别您的选择，请回复'继续'无酒店规划，或'重新生成'重新选择目的地/时间"
        }
    }


def build_alternatives_prompt(interaction: dict) -> dict:
    """有替代酒店但用户选择无法识别时的提示"""
    alternatives = interaction.get("alternatives", [])
    alt_list = "\n".join([
        f"{i+1}. {h['hotel_name']} - ¥{h.get('price', '?')}/晚"
        for i, h in enumerate(alternatives)
    ])
    return {
        "interaction": {
            **interaction,
            "message": f"无法识别您的选择。\n可选：\n{alt_list}\n\n或回复'换一家'/'继续无酒店'/'重新生成'"
        }
    }


def build_alarm_prompt() -> dict:
    """闹钟设置选择时的提示"""
    return {
        "interaction": {
            "message": "无法识别您的选择，请回复'是'设置闹钟，或'否'不需要"
        }
    }
```

### 6.4 Handler A：handle_hotel_selection（等待用户选酒店）

```
状态：waiting_user_hotel
来源：researcher 查到有空房
前置条件：interaction.hotels = [酒店列表]
```

```python
def handle_hotel_selection(state, user_msg, interaction):
    hotels = interaction.get("hotels", [])
    user_msg_clean = user_msg.strip()

    # ── 情况1：用户选择了某个酒店 ──
    chosen = parse_hotel_choice(user_msg, hotels)
    if chosen:
        return {
            "interaction": {
                **interaction,
                "chosen": chosen,
                "status": "confirmed",
            }
        }

    # ── 情况2：用户选择不要酒店 ──
    if user_msg_clean in ("不需要", "不要酒店", "跳过", "继续"):
        return {
            "interaction": {
                **interaction,
                "chosen": None,
                "status": "confirmed",
            }
        }

    # ── 情况3：无法识别，返回提示 ──
    return build_hotel_selection_prompt(hotels)
```

**处理后 flow**：

```
handle_hotel_selection
        │
        ├── 选中酒店 → interaction.chosen = 酒店对象, status = "confirmed"
        │               ↓
        │           supervisor 下一步 → planner
        │
        ├── 不要酒店 → interaction.chosen = None, status = "confirmed"
        │               ↓
        │           supervisor 下一步 → planner（不带酒店）
        │
        └── 无法识别 → 返回提示让用户重新选
                        ↓
                    前端展示提示，等待用户再次输入
```

### 6.5 Handler B：handle_no_hotel_decision（researcher 发现全部没房）

```
状态：waiting_user_decision
来源：researcher 查到所有酒店都没空房
前置条件：interaction.message = "抱歉，XX地区7月暂无空房"
```

```python
def handle_no_hotel_decision(state, user_msg, interaction):
    user_msg_clean = user_msg.strip()

    # ── 情况1：用户选择无酒店继续 ──
    if user_msg_clean in ("继续", "不需要酒店", "不要酒店", "不用了", "跳过"):
        return {
            "interaction": {
                "type": "hotel_select",
                "status": "confirmed",
                "chosen": None,
            },
            "booking_status": "skipped"
        }

    # ── 情况2：用户选择重新生成 ──
    elif user_msg_clean in ("重新生成", "重新规划", "重新安排"):
        # needs_replan 触发 supervisor 清空 state 重新开始
        return {
            "needs_replan": True,
            "interaction": None,
        }

    # ── 情况3：无法识别，返回提示 ──
    return build_no_hotel_prompt()
```

**处理后 flow**：

```
handle_no_hotel_decision
        │
        ├── 继续 → interaction.chosen = None, status = "confirmed"
        │          booking_status = "skipped"
        │               ↓
        │           supervisor 下一步 → planner（无酒店）
        │
        └── 重新生成 → needs_replan = True
                       interaction = None
                       ↓
                   supervisor → 清空 state → task_analyzer（重新开始）
```

```
  - interaction.chosen = 用户原来选的酒店
```

```python
def handle_hotel_alternatives(state, user_msg, interaction):
    user_msg_clean = user_msg.strip()
    alternatives = interaction.get("alternatives", [])

    # ── 情况1：用户选择"换一家" ──
    if user_msg_clean in ("换一家", "看看其他的", "换一个", "其他"):
        if alternatives:
            return {
                "interaction": {
                    **interaction,
                    "type": "hotel_select",
                    "status": "waiting_user_hotel",
                    "hotels": alternatives,
                    "chosen": None,
                    "alternatives": None,
                }
            }

    # ── 情况2：用户选择无酒店继续 ──
    if user_msg_clean in ("继续", "不需要酒店", "不要酒店", "不用了", "跳过"):
        return {
            "interaction": {
                "type": "hotel_select",
                "status": "confirmed",
                "chosen": None,
            },
            "booking_status": "skipped"
        }

    # ── 情况3：用户选择重新生成 ──
    if user_msg_clean in ("重新生成", "重新规划"):
        return {
            "needs_replan": True,
            "interaction": None,
        }

    # ── 情况4：用户直接选择了某个替代酒店 ──
    chosen = parse_hotel_choice(user_msg, alternatives)
    if chosen:
        return {
            "interaction": {
                **interaction,
                "chosen": chosen,
                "status": "confirmed",
            }
        }

    # ── 情况5：无法识别 ──
    return build_alternatives_prompt(interaction)
```

**处理后 flow**：

```

        │
        ├── "换一家" → 切换到 waiting_user_hotel
        │              interaction.hotels = alternatives
```

### 6.8 闹钟设置时机

闹钟不是在 handler 里直接设置的，而是在 **flow 正常结束后**（到达 END 前的 parse_plan 阶段）检查 `interaction.alarm_set` 来设置：

```python
# multi_agent_travel.py 出口处

def check_and_set_alarm(result):
    interaction = result.get("interaction", {})

    if interaction.get("alarm_set"):
        task = result.get("task")
        # 计算退房日期作为闹钟触发时间（提前1小时提醒）
        alarm_time = calculate_alarm_time(task.checkout)

        CronCreate(
            cron=f"0 {alarm_time.hour} {alarm_time.day} * *",
            prompt=f"您关注的 {task.destination} 酒店可能有空房了！",
            recurring=False,
        )
        logger.info(f"已设置闹钟：{alarm_time}")


def calculate_alarm_time(checkout_date: str) -> datetime:
    """计算闹钟触发时间：退房日前1小时"""
    from datetime import datetime, timedelta
    checkout = datetime.strptime(checkout_date, "%Y-%m-%d")
    # 退房时间通常是中午12点，提前1小时 = 11点
    alarm = checkout.replace(hour=11, minute=0) - timedelta(hours=1)
    return alarm
```

## 七、Confirm_and_Book 节点

```python
async def confirm_and_book_node(state):
    task = state.get("task")
    chosen = state.get("interaction", {}).get("chosen")

    # 用户选择不要酒店
    if not chosen:
        return {"booking_status": "skipped"}

    # 1. 再次查询房间是否还有（防止 T1→T3 这段时间被抢走）
    current_availability = await call_java_hotel_api(
        hotel_id=chosen["hotel_id"],
        checkin=task.start_date,
        checkout=task.end_date,
    )

    if not current_availability.get("has_rooms"):
        alternatives = current_availability.get("similar_hotels", [])

        if alternatives:
            # 2a. 房间没了，有替代酒店
            return {
                "interaction": {
                    "type": "hotel_rebooking",
                    "status": "waiting_user_alternatives",
                    "message": f"抱歉，您选择的 {chosen['hotel_name']} 刚才被订走了",
                    "chosen": chosen,
                    "alternatives": alternatives,
                }
            }
        else:
            # 2b. 房间没了，无替代酒店
            return {
                "interaction": {
                    "type": "hotel_rebooking",
                    "status": "waiting_user_alarm",
                    "message": f"抱歉，您选择的 {chosen['hotel_name']} 刚才被订走，且附近无其他空房",
                }
            }

    # 3. 房间还有 → 正式预定
    booking_result = await call_java_hotel_book(
        hotel_id=chosen["hotel_id"],
        user_id=task.user_id,
        checkin=task.start_date,
        checkout=task.end_date,
    )

    if booking_result.get("success"):
        return {
            "booking_status": "confirmed",
            "booking": booking_result,
            "interaction": {
                **state.get("interaction", {}),
                "status": "confirmed",
            },
        }
    else:
        # 4. 预定失败
        return {
            "interaction": {
                "type": "hotel_rebooking",
                "status": "waiting_user_alarm",
                "message": f"预定失败: {booking_result.get('error')}",
            }
        }
```

```python
def save_flow_state(session_id: int, flow_id: str, state: dict):
    """保存 flow 中间状态到 Redis"""
    key = f"agent:state:{session_id}:{flow_id}"
    redis_client.set(key, json.dumps(state, ensure_ascii=False), ex=3600)  # 1小时过期


def get_flow_state(session_id: int, flow_id: str) -> dict | None:
    """从 Redis 恢复 flow 状态"""
    key = f"agent:state:{session_id}:{flow_id}"
    data = redis_client.get(key)
    return json.loads(data) if data else None


def clear_flow_state(session_id: int, flow_id: str):
    """清除 flow 状态"""
    key = f"agent:state:{session_id}:{flow_id}"
    redis_client.delete(key)
```
