### 1.4 地图显示核心逻辑

```javascript
// handleSSEEvent() 中的 agent_complete 处理
case 'agent_complete': {
    const data = event.data;
    
    // 1. 显示聊天内容
    addMessage('assistant', data.content);
    
    // 2. 如果是旅游规划(plan)类型，提取景点
    if (data.type === 'plan') {
        // 从 parsed_plan.daily_plans 中提取所有地点
        const locationSet = new Set();
        data.parsed_plan.daily_plans.forEach(dayPlan => {
            dayPlan.activities.forEach(activity => {
                // activity 可能是字符串或对象
                const loc = typeof activity === 'string' 
                    ? activity 
                    : activity.location;
                if (loc) locationSet.add(loc);
            });
        });
        
        // 3. 保存并显示
        currentDestinations = Array.from(locationSet);
        showDestinationsOnMap(currentDestinations);  // 地图打点
        updateDestList(currentDestinations);          // 更新列表
    }
}
```

### 1.5 地图打点流程 (showDestinationsOnMap)

```javascript
function showDestinationsOnMap(destinations) {
    // 1. 清除旧的标记和路线
    markers.forEach(m => m.setMap(null));
    markers = [];
    if (routeLine) routeLine.setMap(null);
    
    // 2. 遍历每个目的地，调用后端获取坐标
    destinations.forEach((dest, index) => {
        fetch(`/api/route/geocode?address=${encodeURIComponent(dest)}`)
            .then(r => r.json())
            .then(data => {
                if (data.success) {
                    // 3. 创建标记点
                    const pos = new AMap.LngLat(data.lng, data.lat);
                    const marker = new AMap.Marker({
                        position: pos,
                        title: dest,           // 鼠标悬停显示
                        label: {
                            content: (index + 1) + '',  // 显示序号
                            config: { position: 'TOP' }
                        }
                    });
                    marker.setMap(map);
                    markers.push(marker);
                }
            });
    });
    
    // 4. 调整视野显示全部
    map.setFitView();
}
```

### 1.6 路线规划流程 (planRoute)

```javascript
async function planRoute() {
    // 1. 获取用户输入的起点
    const startAddr = startInput.value.trim();
    
    // 2. 调用后端API获取完整路线
    const res = await fetch('/api/route/map', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
            start_address: startAddr,           // 起点
            destinations: currentDestinations, // 所有景点
            travel_mode: 'driving'             // 驾车
        })
    });
    const data = await res.json();
    
    // 3. 绘制路线
    if (data.polyline_points) {
        const path = data.polyline_points.map(
            p => new AMap.LngLat(p[0], p[1])
        );
        routeLine = new AMap.Polyline({
            path: path,                    // 路线坐标点
            strokeColor: '#3b82f6',       // 蓝色
            strokeWeight: 5,               // 线宽5px
        });
        routeLine.setMap(map);
    }
    
    // 4. 显示距离和时间
    document.getElementById('routeSummary').innerHTML = `
        路线总距离：${data.total_distance / 1000} 公里
        预计时间：约${data.total_duration / 60} 分钟
    `;
}
```

### 2.2 地址解析接口

@router.get("/geocode")
async def geocode_address(address: str):
    """
    将地址转换为经纬度坐标
    
    前端调用: GET /api/route/geocode?address=广州塔
    返回: {"success": true, "lng": 113.327, "lat": 23.130, ...}
    """
    gaode = get_gaode_map_tools()
    result = gaode.address_to_coord(address)
    
    if result:
        return {"success": True, **result}
    return {"success": False, "error": "解析失败"}
```

class GaodeMapTools:
    """高德地图工具类"""
    
    def address_to_coord(self, address):
        """
        地址 → 坐标
        
        使用高德地理编码API: https://restapi.amap.com/v3/geocode/geo
        
        参数: address = "广州市天河区广州塔"
        返回: {"lng": 113.327, "lat": 23.130, "province": "广东省", ...}
        """
        
    def get_full_route_with_waypoints(self, start_address, destinations, travel_mode):
        """
        获取完整路线（核心方法）
        
        流程:
        1. 解析起始地址 → 坐标
        2. 拆分合并目的地（如 "故宫/颐和园" → ["故宫", "颐和园"]）
        3. 遍历每个目的地 → 坐标
        4. 逐段调用驾车路线API
        5. 收集所有拐点坐标
        6. 返回完整路线信息
        """
```

## 五、关键数据结构

### 5.1 TravelPlan 结构

```python
@dataclass
class TravelPlan:
    title: str                    # "上海两天游"
    destination: str               # "上海"
    daily_plans: List[DailyActivity]  # 每天的计划

@dataclass
class DailyActivity:
    day: int                       # 1
    theme: str                     # "上午博物馆，下午外滩"
    activities: List[str]          # ["上海博物馆", "外滩"]
    location: str                  # "上海市中心"  (← 问题：这里经常为空)
```

### 5.2 高德API返回格式

```json
// 地址解析返回
{
    "status": "1",
    "geocodes": [{
        "location": "113.327,23.130",  // "经度,纬度"
        "province": "广东省",
        "city": "广州市",
        "district": "海珠区"
    }]
}

// 驾车路线返回
{
    "status": "1",
    "route": {
        "paths": [{
            "distance": 15000,      // 距离(米)
            "duration": 1800,        // 时间(秒)
            "steps": [{
                "instruction": "向东行驶",
                "distance": 500,
                "polyline": "113.327,23.130;113.328,23.131;..."  // 拐点坐标
            }]
        }]
    }
}
```

## 六、问题排查

| 问题 | 可能原因 | 解决方案 |
|------|----------|----------|
| 地图没显示目的地 | `daily_plans[].location`为空 | 改从`activities[].location`提取 |
| 地址解析失败 | 目的地含特殊字符 | 用`/`拆分后重试 |
| 路线不准确 | 多地点合并无法解析 | 正确拆分合并目的地 |
| MCP未连接 | RAG服务未启动 | 启动qdrant_rag_server |
